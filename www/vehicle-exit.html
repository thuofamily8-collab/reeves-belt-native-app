🚛

Loading vehicles...

'; RBApi.getVehiclesInside().then(function (res) { var vehicles = (res && res.vehicles) || (res.data && res.data.vehicles) || []; renderVehicles(vehicles); }).catch(function (err) { container.innerHTML = '
' + '
📡
' + '

Cannot load vehicles

' + '

' + (err.error || 'Network error') + '

' + '
'; }); } // ============================================================ // RENDER VEHICLES // ============================================================ function renderVehicles(vehicles) { var container = document.getElementById('vehiclesList'); if (!container) return; if (!vehicles || vehicles.length === 0) { container.innerHTML = '
' + '
🚛
' + '

No vehicles currently inside

' + '
'; return; } var html = ''; for (var i = 0; i < vehicles.length; i++) { var v = vehicles[i]; var entryTime = v.entry_time ? formatTime(v.entry_time) : 'Just now'; // Pre-weighed badge (only if this vehicle came from a pre-weighed truck) var preweighedBadge = v.preweighed_code ? '
📥 PRE-WEIGHED · ' + escapeHtml(v.preweighed_code) + '
' : ''; html += '
' + '
' + escapeHtml(v.plate_number) + '
' + '
🚚 ' + escapeHtml(v.vehicle_type || '—') + (v.vehicle_model ? ' · ' + escapeHtml(v.vehicle_model) : '') + '
' + '
👤 ' + escapeHtml(v.driver_name || '—') + '
' + '
📞 ' + escapeHtml(v.driver_phone || '—') + '
' + '
🕐 In: ' + entryTime + '
' + (v.entry_weight_kg ? '
⚖️ Entry: ' + v.entry_weight_kg + ' kg
' : '') + preweighedBadge + '
'; } container.innerHTML = html; } // ============================================================ // START EXIT PROCESS // ============================================================ var _exitVehicleId = null; function startExit(logId, plate) { _exitVehicleId = logId; var modal = document.getElementById('exitModal'); if (!modal) { // Fallback: use prompt var w = prompt('Enter exit weight for ' + plate + ' (kg):'); if (w === null) return; var c = prompt('Exit comment (optional):') || ''; submitExit(logId, parseFloat(w), c); return; } // Show modal var titleEl = document.getElementById('exitModalTitle'); if (titleEl) titleEl.textContent = 'Exit · ' + plate; var weightInput = document.getElementById('exitWeight'); if (weightInput) weightInput.value = ''; var commentInput = document.getElementById('exitComment'); if (commentInput) commentInput.value = ''; modal.classList.add('show'); } function cancelExit() { var modal = document.getElementById('exitModal'); if (modal) modal.classList.remove('show'); _exitVehicleId = null; } function confirmExit() { if (!_exitVehicleId) return; var weightInput = document.getElementById('exitWeight'); var commentInput = document.getElementById('exitComment'); var exitWeight = weightInput ? parseFloat(weightInput.value) : null; var exitComment = commentInput ? (commentInput.value || '').trim() : ''; if (!exitWeight || exitWeight <= 0) { if (typeof RBApp !== 'undefined' && RBApp.showToast) { RBApp.showToast('Enter a valid exit weight', 'error'); } else { alert('Enter a valid exit weight'); } return; } submitExit(_exitVehicleId, exitWeight, exitComment); } function submitExit(logId, exitWeight, exitComment) { var btn = document.getElementById('exitConfirmBtn'); if (btn) { btn.disabled = true; btn.textContent = 'Processing...'; } RBApi.processVehicleExit(logId, exitWeight, exitComment).then(function (res) { if (navigator.vibrate) navigator.vibrate(150); var data = res || {}; var netW = data.net_weight_kg !== undefined ? data.net_weight_kg : (data.data && data.data.net_weight_kg); var msg = '✅ Exit recorded'; if (netW !== null && netW !== undefined) { msg += ' · Net ' + netW + ' kg'; } if (typeof RBApp !== 'undefined' && RBApp.showToast) { RBApp.showToast(msg, 'success'); } else { alert(msg); } cancelExit(); _exitVehicleId = null; if (btn) { btn.disabled = false; btn.textContent = 'Confirm Exit'; } // Refresh the list setTimeout(function () { loadVehiclesInside(); }, 500); }).catch(function (err) { var msg = err && (err.error || err.message) ? (err.error || err.message) : 'network error'; if (typeof RBApp !== 'undefined' && RBApp.showToast) { RBApp.showToast('Failed: ' + msg, 'error'); } else { alert('Failed: ' + msg); } if (btn) { btn.disabled = false; btn.textContent = 'Confirm Exit'; } }); } // ============================================================ // HELPERS // ============================================================ function formatTime(dateStr) { try { var d = new Date(dateStr.replace(' ', 'T')); if (isNaN(d.getTime())) return dateStr; var now = new Date(); var diff = Math.floor((now - d) / 1000); if (diff < 60) return 'Just now'; if (diff < 3600) return Math.floor(diff / 60) + 'm ago'; if (diff < 86400) return Math.floor(diff / 3600) + 'h ago'; return d.getHours().toString().padStart(2, '0') + ':' + d.getMinutes().toString().padStart(2, '0'); } catch (e) { return dateStr; } } function escapeHtml(text) { if (!text) return ''; var div = document.createElement('div'); div.textContent = text; return div.innerHTML; } // ============================================================ // INIT // ============================================================ document.addEventListener('DOMContentLoaded', function () { if (!RBAuth.requireLogin()) return; loadVehiclesInside(); // Refresh every 30s setInterval(function () { if (!document.hidden) loadVehiclesInside(); }, 30000); });
